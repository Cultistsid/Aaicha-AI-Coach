### Task 1: `useStickyScroll` hook

**Files:**
- Create: `src/lib/useStickyScroll.ts`

**Interfaces:**
- Produces: `useStickyScroll<T extends HTMLElement>(dep: unknown): { ref: (node: T | null) => void; pinned: boolean; jumpToLatest: () => void }`. Tasks 2 and 5 attach `ref` to scroll containers and render a pill when `!pinned`.

- [ ] **Step 1: Write the hook**

```ts
import { useCallback, useEffect, useRef, useState } from "react"

const PIN_THRESHOLD_PX = 48

// Keeps a scroll pane following its bottom edge only while the user is
// already there. Scrolling up unpins; `jumpToLatest` re-pins. `dep` is any
// value that changes when content grows (combined text length works).
// Callback-ref so remounts (overlay flip) re-attach the scroll listener.
export function useStickyScroll<T extends HTMLElement>(dep: unknown) {
  const [el, setEl] = useState<T | null>(null)
  const [pinned, setPinned] = useState(true)
  const pinnedRef = useRef(true)

  const ref = useCallback((node: T | null) => setEl(node), [])

  useEffect(() => {
    if (!el) return
    const onScroll = () => {
      const atBottom =
        el.scrollHeight - el.scrollTop - el.clientHeight <= PIN_THRESHOLD_PX
      pinnedRef.current = atBottom
      setPinned(atBottom)
    }
    el.addEventListener("scroll", onScroll, { passive: true })
    // Fresh mount starts pinned at the bottom.
    pinnedRef.current = true
    setPinned(true)
    el.scrollTop = el.scrollHeight
    return () => el.removeEventListener("scroll", onScroll)
  }, [el])

  // Follow growth only while pinned. Instant, never smooth — queued smooth
  // animations are what caused the yank during token streaming.
  useEffect(() => {
    if (el && pinnedRef.current) el.scrollTop = el.scrollHeight
  }, [dep, el])

  const jumpToLatest = useCallback(() => {
    if (!el) return
    el.scrollTop = el.scrollHeight
    pinnedRef.current = true
    setPinned(true)
  }, [el])

  return { ref, pinned, jumpToLatest }
}
```

### Task 6: Dev shim so the renderer runs in a plain browser

**Files:**
- Create: `src/lib/devElectronShim.ts`
- Modify: `src/main.tsx`

Purpose: `window.electronAPI` doesn't exist outside Electron, so the renderer can't be QA-driven in Chrome. A dev-only shim fakes STT events and streamed coach turns. Loaded ONLY when `import.meta.env.DEV && !window.electronAPI` — never in the packaged app (electronAPI exists) and never in prod builds.

- [ ] **Step 1: Write the shim**

```ts
// Dev-only stand-in for the Electron preload API so the renderer can run in
// a plain browser for UI QA. Streams canned transcript + coach turns.
type Handler = (payload: any) => void

const sttHandlers = new Set<Handler>()
const coachDelta = new Set<Handler>()
const coachDone = new Set<Handler>()

const SCRIPT = [
  "Tell me about a time you had to clean really messy data.",
  "Walk me through how you'd design an orders API.",
  "Why do you want to work here?"
]

const COACH_REPLY = [
  "**Answer:** I'd lead with the corrupt `ordered_by` rows. ",
  "Six RECEIVED rows named a different person entirely — ",
  "the fix was reading the field only from the PENDING row, ",
  "so corruption never even needs detecting.\n\n",
  "```python\nby = group[group.status == 'PENDING'].iloc[0].ordered_by\n",
  "# only the authoritative row — corrupt rows are ignored by construction\n```\n",
  "Close with the mutation-testing story if they push on QA."
]

let sttTimer: number | null = null

const shim = {
  sttStart: async () => {
    sttHandlers.forEach((h) => h({ kind: "status", status: "open" }))
    let i = 0
    sttTimer = window.setInterval(() => {
      const line = SCRIPT[i % SCRIPT.length]
      sttHandlers.forEach((h) => h({ kind: "interim", text: line.slice(0, 20) }))
      window.setTimeout(
        () => sttHandlers.forEach((h) => h({ kind: "final", text: line })),
        700
      )
      i++
    }, 6000)
    return { success: true }
  },
  sttStop: async () => {
    if (sttTimer) window.clearInterval(sttTimer)
    sttHandlers.forEach((h) => h({ kind: "status", status: "closed" }))
    return { success: true }
  },
  sttSendAudio: (_b64: string) => {},
  onSTTEvent: (h: Handler) => {
    sttHandlers.add(h)
    return () => sttHandlers.delete(h)
  },
  coachReset: async () => ({ success: true }),
  coachTurn: async (turnId: string, _text: string, _img?: string) => {
    let k = 0
    const t = window.setInterval(() => {
      if (k < COACH_REPLY.length) {
        coachDelta.forEach((h) => h({ turnId, text: COACH_REPLY[k] }))
        k++
      } else {
        window.clearInterval(t)
        coachDone.forEach((h) => h({ turnId, text: COACH_REPLY.join("") }))
      }
    }, 180)
    return { success: true }
  },
  onCoachDelta: (h: Handler) => {
    coachDelta.add(h)
    return () => coachDelta.delete(h)
  },
  onCoachDone: (h: Handler) => {
    coachDone.add(h)
    return () => coachDone.delete(h)
  },
  onCoachError: (_h: Handler) => () => {},
  onCoachScreenAttached: (_h: Handler) => () => {},
  getOverlay: async () => ({ active: false }),
  setOverlay: async (_a: boolean) => ({ active: false }),
  toggleOverlay: async () => ({ active: false })
}

export function installDevShim() {
  ;(window as any).electronAPI = new Proxy(shim, {
    // Any API the shim doesn't model becomes a harmless no-op instead of a
    // crash, so Chat/Setup screens still render.
    get(target, prop) {
      if (prop in target) return (target as any)[prop]
      return async () => undefined
    }
  })
}
```
